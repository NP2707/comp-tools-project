GitSkills similarity study

Research question: what families of similar agent instructions appear beyond identical byte contents?

Download code, notebook, results, and reproduction instructions

Dataset used: the GitHub sample. The full Zenodo dataset was not downloaded or analyzed. The sample contains 29,786 file occurrences and 13,000 distinct contents; the default filters retain 11,015 contents.

Results and interpretation

LSH proposed 3,512 of 60,659,605 possible pairs. The complete run took 54.8 seconds including exhaustive evaluation and comparisons; peak process memory was 2,417 MiB. The baseline's nine correctness tests passed and its notebook was executed.

Some high-similarity pairs share generic templates despite different purposes. At resolution 1, Louvain reproduces the connected components at each evaluated threshold. Human review and a stronger graph formulation are still needed. Subset recall is not a guarantee of complete full-corpus recall. Verified precision follows the lexical threshold definition, not human judgments of useful similarity.

The template audit checks this interpretation: one generic family accounts for 92.7% of pairs crossing the provisional purpose clusters. Shared text and cluster assignments alone do not establish shared functionality.

def sha256(path):
    digest = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            digest.update(block)
    return digest.hexdigest()


def write_csv(path, rows, fields):
    with path.open("w", newline="", encoding="utf-8") as stream:
        writer = csv.DictWriter(stream, fieldnames=fields)
        writer.writeheader()
        writer.writerows(rows)


def plot_results(output, evaluation, groups, edges):
    os.environ.setdefault("MPLCONFIGDIR", str(ROOT / ".cache" / "matplotlib"))
    import matplotlib
    matplotlib.use("Agg")
    import matplotlib.pyplot as plt
    fig, axes = plt.subplots(1, 3, figsize=(13, 3.8), constrained_layout=True)
    for bands in (16, 32, 64):
        values = [row for row in evaluation if row["bands"] == bands and row["recall"] is not None]
        axes[0].plot([r["threshold"] for r in values], [r["recall"] for r in values],
                     marker="o", label=f"{bands} bands × {128 // bands} rows")
    axes[0].set(xlabel="Exact Jaccard threshold", ylabel="Recall on exhaustive subset", ylim=(0, 1.05))
    axes[0].legend(fontsize=8)
    sizes = sorted((len(g) for g in groups if len(g) > 1), reverse=True)
    axes[1].plot(range(1, len(sizes) + 1), sizes, marker=".", linewidth=1)
    axes[1].set(xlabel="Community rank (isolates omitted)", ylabel="Distinct contents", yscale="log")
    axes[2].hist(list(edges.values()), bins=20, color="#277a87")
    axes[2].set(xlabel="Verified edge Jaccard", ylabel="Pairs")
    fig.suptitle("GitSkills sample: retrieval and similarity communities")
    for suffix in ("png", "svg"):
        fig.savefig(output / f"overview.{suffix}", dpi=160)
    plt.close(fig)


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--database", type=Path, default=ROOT / "data/raw/agent_skills_sample.db")
    parser.add_argument("--output", type=Path, default=ROOT / "results/sample")
    parser.add_argument("--limit", type=int, default=0, help="0 = all eligible sample contents")
    parser.add_argument("--seed", type=int, default=42)
    parser.add_argument("--shingle-size", type=int, default=5)
    parser.add_argument("--min-tokens", type=int, default=30)
    parser.add_argument("--threshold", type=float, default=0.8)
    parser.add_argument("--bands", type=int, default=32)
    parser.add_argument("--rows", type=int, default=4)
    parser.add_argument("--max-candidates", type=int, default=2_000_000)
    parser.add_argument("--eval-size", type=int, default=4000)
    parser.add_argument("--lsa-components", type=int, default=64)
    parser.add_argument("--lsa-neighbors", type=int, default=5)
    parser.add_argument("--lsa-threshold", type=float, default=0.8)
    parser.add_argument("--include-nonstandard", action="store_true",
                        help="Disable filename and frontmatter filters for a sensitivity run")
    parser.add_argument("--keep-frontmatter", action="store_true")
    parser.add_argument("--skip-lsa", action="store_true")
    args = parser.parse_args(argv)
    if (args.limit < 0 or args.shingle_size < 1 or args.min_tokens < args.shingle_size
            or args.eval_size < 2 or args.bands < 1 or args.rows < 1 or args.bands * args.rows != 128
            or not 0 < args.threshold <= 1 or args.max_candidates < 1
            or args.lsa_components < 1 or args.lsa_neighbors < 1 or not 0 < args.lsa_threshold <= 1):
        parser.error("Check positive sizes, thresholds in (0,1], min_tokens >= shingle_size, and bands*rows=128")
    if args.output.exists() and any(args.output.iterdir()):
        parser.error("Output directory is not empty; choose a new directory to preserve previous runs")
    args.output.mkdir(parents=True, exist_ok=True)
    start = time.perf_counter()
    timings = {}

    def stage(name):
        print(name, flush=True)
        return time.perf_counter()

    t = stage("Loading and verifying distinct contents...")
    skills, corpus_stats = load_corpus(args.database, limit=args.limit, seed=args.seed,
                                     min_tokens=args.min_tokens, strict=not args.include_nonstandard,
                                     strip_frontmatter=not args.keep_frontmatter)
    if len(skills) < 3:
        raise ValueError("At least three eligible documents are required")
    documents = [shingles(skill.tokens, args.shingle_size) for skill in skills]
    timings["load_and_shingle"] = time.perf_counter() - t
    t = stage(f"Computing MinHash signatures for {len(skills):,} contents...")
    matrix = signatures(documents, 128, args.seed)
    timings["minhash"] = time.perf_counter() - t
    t = stage("Generating LSH candidates and verifying exact similarities...")
    candidates = lsh_candidates(matrix, args.bands, args.rows, args.max_candidates)
    timings["lsh_candidates"] = time.perf_counter() - t
    t = time.perf_counter()
    thresholds = sorted({0.6, 0.7, 0.8, 0.9, args.threshold})
    scores = verified_edges(documents, sorted(candidates), min(thresholds))
    edges = {pair: score for pair, score in scores.items() if score >= args.threshold}
    timings["exact_verification"] = time.perf_counter() - t
    t = stage("Evaluating retrieval against exhaustive comparisons...")
    evaluation, indices = evaluate_retrieval(documents, size=args.eval_size, seed=args.seed,
                                             thresholds=thresholds)
    timings["evaluation"] = time.perf_counter() - t
    write_csv(args.output / "retrieval.csv", evaluation, list(evaluation[0]))
    (args.output / "evaluation_ids.json").write_text(json.dumps([skills[i].sha for i in indices], indent=2) + "\n")
    t = stage("Clustering the weighted similarity graph...")
    graph, groups, graph_stats = cluster_graph(len(skills), edges, seed=args.seed)
    sensitivity = []
    for threshold in thresholds:
        subset = {pair: score for pair, score in scores.items() if score >= threshold}
        for resolution in (0.5, 1.0, 2.0):
            _, _, stats = cluster_graph(len(skills), subset, resolution=resolution, seed=args.seed)
            sensitivity.append({"threshold": threshold, "resolution": resolution, **stats})
    write_csv(args.output / "graph_sensitivity.csv", sensitivity, list(sensitivity[0]))
    timings["graph"] = time.perf_counter() - t
    membership = {node: community for community, nodes in enumerate(groups) for node in nodes}
    nodes = [{"sha": skill.sha, "name": skill.name, "representative_repo": skill.repo,
              "representative_path": skill.path, "occurrences": skill.copies,
              "repositories": len(skill.repos), "tokens": len(skill.tokens),

             for i, skill in enumerate(skills)]
    write_csv(args.output / "nodes.csv", nodes, list(nodes[0]))
    write_csv(args.output / "edges.csv",
              ({"left_sha": skills[i].sha, "right_sha": skills[j].sha, "jaccard": score}
               for (i, j), score in sorted(edges.items())), ["left_sha", "right_sha", "jaccard"])
    communities = [{"community": i, "distinct_contents": len(group),

    write_csv(args.output / "lsa_pairs.csv",
              ({"left_sha": skills[i].sha, "right_sha": skills[j].sha, "cosine": score}
               for (i, j), score in sorted(semantic.items())), ["left_sha", "right_sha", "cosine"])

"""Read distinct contents while retaining occurrence counts for reuse analysis."""
from __future__ import annotations

from dataclasses import dataclass
import hashlib
from pathlib import Path
import random
import re
import sqlite3
import unicodedata


@dataclass(frozen=True)
class Skill:
    sha: str
    name: str
    repo: str
    path: str
    text: str
    tokens: tuple[str, ...]
    copies: int
    repos: tuple[str, ...]


def body_tokens(text: str, strip_frontmatter: bool = True) -> tuple[str, ...]:
    """Normalize Unicode/case; retain words, numbers, underscores and code tokens.

    Remove only a delimited YAML block at the beginning. Keep Markdown body and
    code examples; no stopword removal, stemming, or undocumented truncation.
    """
    text = text.lstrip("\ufeff")
    if strip_frontmatter:
        text = re.sub(r"\A---[^\S\r\n]*\r?\n.*?\r?\n(?:---|\.\.\.)[^\S\r\n]*(?:\r?\n|\Z)",
                      "", text, count=1, flags=re.DOTALL)
    return tuple(re.findall(r"\w+", unicodedata.normalize("NFKC", text).casefold()))


def load_corpus(database: Path, *, limit: int = 0, seed: int = 42,
                min_tokens: int = 30, strict: bool = True,
                strip_frontmatter: bool = True) -> tuple[list[Skill], dict]:
    if limit < 0 or min_tokens < 1:
        raise ValueError("limit must be nonnegative and min_tokens positive")
    if not database.is_file():
        raise FileNotFoundError(f"Missing {database}; run scripts/download_sample.py first")
    connection = sqlite3.connect(database.resolve().as_uri() + "?mode=ro", uri=True)
    connection.row_factory = sqlite3.Row
    with connection:
        required = {"file_sha", "dedup_primary", "content", "name", "description",
                    "frontmatter_valid", "filename", "repo_full_name", "path"}
        columns = {r[1] for r in connection.execute("PRAGMA table_info(artifacts)")}
        if not required <= columns:
            raise ValueError(f"Missing artifact columns: {sorted(required - columns)}")
        occurrences: dict[str, list] = {}
        for row in connection.execute("SELECT file_sha, repo_full_name, filename FROM artifacts"):
            group = occurrences.setdefault(row[0], [0, set(), False])
            group[0] += 1
            group[1].add(row[1])
            group[2] |= row[2] == "SKILL.md"
        stats = {"occurrences": sum(g[0] for g in occurrences.values()),
                 "distinct_contents": len(occurrences), "representatives": 0,
                 "excluded_metadata": 0, "excluded_short": 0, "excluded_missing_text": 0}
        skills, seen = [], set()
        for row in connection.execute("SELECT * FROM artifacts WHERE dedup_primary=1 ORDER BY file_sha"):
            stats["representatives"] += 1
            sha, text = row["file_sha"], row["content"]
            if sha in seen:
                raise ValueError(f"Multiple primary rows for {sha}")
            seen.add(sha)
            if not isinstance(text, str) or not text.strip():
                stats["excluded_missing_text"] += 1
                continue
            raw = text.encode("utf-8")
            actual_sha = hashlib.sha1(f"blob {len(raw)}\0".encode() + raw).hexdigest()
            if actual_sha != sha:
                raise ValueError(f"Content hash mismatch for {sha}")
            copies, repos, has_skill_filename = occurrences[sha]
            if strict and (not has_skill_filename or row["frontmatter_valid"] != 1
                           or not (row["name"] or "").strip()
                           or not (row["description"] or "").strip()):
                stats["excluded_metadata"] += 1
                continue
            tokens = body_tokens(text, strip_frontmatter)
            if len(tokens) < min_tokens:
                stats["excluded_short"] += 1
                continue
            skills.append(Skill(sha, row["name"] or "", row["repo_full_name"],
                                row["path"], text, tokens, copies, tuple(sorted(repos))))
        if seen != set(occurrences):
            raise ValueError("Some content groups have no primary row")
    connection.close()
    stats["eligible_contents"] = len(skills)
    if limit and len(skills) > limit:
        skills = random.Random(seed).sample(skills, limit)
    skills.sort(key=lambda skill: skill.sha)
    stats["selected_contents"] = len(skills)
    stats["selected_occurrences"] = sum(s.copies for s in skills)
    return skills, stats

"""Evaluation and graph/latent-semantic comparisons for the sample study."""
from __future__ import annotations

from itertools import combinations
import random
import time

import networkx as nx
import numpy as np
from sklearn.decomposition import TruncatedSVD
from sklearn.feature_extraction.text import TfidfVectorizer
from sklearn.preprocessing import normalize

from .similarity import jaccard, lsh_candidates, retrieval_metrics, signatures, verified_edges


def evaluate_retrieval(documents, *, size=600, seed=42,
                       thresholds=(0.6, 0.7, 0.8, 0.9)):
    """Independently sample documents, then exhaustively enumerate their pairs.

    Ground truth is defined by exact lexical Jaccard, not human judgments. Never
    draw evaluation pairs from the LSH candidates: doing so would hide misses.
    """
    indices = sorted(random.Random(seed + 1).sample(range(len(documents)), min(size, len(documents))))
    sample = [documents[i] for i in indices]
    start = time.perf_counter()
    truth_scores = verified_edges(sample, combinations(range(len(sample)), 2), min(thresholds))
    exact_seconds = time.perf_counter() - start
    start = time.perf_counter()
    matrix = signatures(sample, 128, seed)
    signature_seconds = time.perf_counter() - start
    rows = []
    for bands, band_rows in [(16, 8), (32, 4), (64, 2)]:
        start = time.perf_counter()
        candidates = lsh_candidates(matrix, bands, band_rows)
        candidate_seconds = time.perf_counter() - start
        start = time.perf_counter()
        verified = verified_edges(sample, sorted(candidates), min(thresholds))
        verify_seconds = time.perf_counter() - start
        for threshold in thresholds:
            truth = {pair for pair, score in truth_scores.items() if score >= threshold}
            selected = {pair for pair, score in verified.items() if score >= threshold}
            rows.append({"documents": len(sample), "bands": bands, "rows": band_rows,
                         "threshold": threshold,
                         **retrieval_metrics(truth, candidates, selected,
                                             len(sample) * (len(sample) - 1) // 2),
                         "exact_seconds_at_min_threshold": exact_seconds,
                         "signature_seconds": signature_seconds,
                         "candidate_seconds": candidate_seconds, "verify_seconds": verify_seconds,
                         "lsh_seconds_at_min_threshold": signature_seconds + candidate_seconds + verify_seconds})
    return rows, indices


def cluster_graph(count, edges, *, resolution=1.0, seed=42):
    """Include isolates, avoid self-loops, and canonicalize community IDs."""
    graph = nx.Graph()
    graph.add_nodes_from(range(count))
    graph.add_weighted_edges_from((i, j, score) for (i, j), score in sorted(edges.items()) if i != j)
    if graph.number_of_edges():
        groups = nx.community.louvain_communities(graph, weight="weight", resolution=resolution, seed=seed)
        modularity = nx.community.modularity(graph, groups, weight="weight", resolution=resolution)
    else:
        groups = [{i} for i in range(count)]
        modularity = None
    groups = sorted((sorted(group) for group in groups), key=lambda group: (-len(group), group[0]))
    stats = {"nodes": count, "edges": graph.number_of_edges(),
             "isolates": nx.number_of_isolates(graph), "communities": len(groups),
             "nontrivial_communities": sum(len(group) > 1 for group in groups),
             "largest_community": max(map(len, groups), default=0),
             "connected_components": nx.number_connected_components(graph), "modularity": modularity}
    return graph, groups, stats


def semantic_pairs(skills, *, components=64, neighbors=5, threshold=0.8, seed=42):
    """TF-IDF -> truncated SVD (LSA) -> normalized cosine nearest neighbours.

    Fit on this corpus for descriptive exploration. This is not a held-out
    prediction experiment or evidence that similar documents are copied.
    """
    if len(skills) < 3:
        return {}, {"status": "skipped: fewer than three documents"}
    vectorizer = TfidfVectorizer(tokenizer=str.split, token_pattern=None, lowercase=False,
                                min_df=2, max_features=30_000, sublinear_tf=True, dtype=np.float32)
    try:
        matrix = vectorizer.fit_transform(" ".join(skill.tokens) for skill in skills)
    except ValueError as error:
        return {}, {"status": f"skipped: {error}"}
    rank = min(components, matrix.shape[0] - 1, matrix.shape[1] - 1)
    if rank < 1:
        return {}, {"status": "skipped: insufficient vocabulary"}
    svd = TruncatedSVD(n_components=rank, n_iter=7, random_state=seed)
    embedded = normalize(svd.fit_transform(matrix))
    k = min(neighbors, len(skills) - 1)
    pairs = {}
    for start in range(0, len(skills), 256):
        scores = embedded[start:start + 256] @ embedded.T
        for offset, row in enumerate(scores):
            i = start + offset
            row[i] = -np.inf
            nearest = np.argpartition(-row, k - 1)[:k]
            for j in sorted(nearest, key=lambda j: (-float(row[j]), int(j))):
                score = float(np.clip(row[j], -1, 1))
                if score >= threshold:
                    pair = tuple(sorted((i, int(j))))
                    pairs[pair] = score
    return pairs, {"status": "complete", "features": matrix.shape[1], "components": rank,
                   "explained_variance_ratio": float(svd.explained_variance_ratio_.sum()),
                   "neighbors": k, "cosine_threshold": threshold, "pairs": len(pairs)}


def review_sample(skills, documents, lexical, semantic, *, per_group=12, seed=42):
    """Stratified review candidates; blank labels must be filled by people."""
    lexical_set, semantic_set = set(lexical), set(semantic)
    strata = {"both": lexical_set & semantic_set,
              "lexical_only": lexical_set - semantic_set,
              "lsa_only": semantic_set - lexical_set}
    rng, rows = random.Random(seed + 2), []
    for stratum, pairs in strata.items():
        for i, j in rng.sample(sorted(pairs), min(per_group, len(pairs))):
            rows.append({"stratum": stratum, "stratum_size": len(pairs),
                         "left_sha": skills[i].sha, "right_sha": skills[j].sha,
                         "left_name": skills[i].name, "right_name": skills[j].name,
                         "jaccard": jaccard(documents[i], documents[j]),
                         "lsa_cosine": semantic.get((i, j)),
                         "reviewer_1_label": "", "reviewer_2_label": "", "notes": ""})
    return rows

STOP_WORDS = sorted(set(ENGLISH_STOP_WORDS) | GENERIC_WORDS)


def purpose_tokens(text):
    """Split identifiers, discard opaque numeric IDs, and preserve Unicode words."""
    text = unicodedata.normalize("NFKC", text).casefold()
    text = re.sub(r"\bu\d+\b", " ", text)
    return re.findall(r"[^\W\d_][^\W_]+", text)


def feature_views(skills, descriptions):
    metadata = [s.name.replace("-", " ").replace("_", " ") + " " + descriptions[s.sha] for s in skills]
    body = [" ".join(s.tokens).replace("_", " ") for s in skills]
    vectorizers, matrices = [], []
    for texts, ngrams, maximum, cutoff in ((metadata, (1, 2), 40_000, 0.7), (body, (1, 1), 30_000, 0.4)):
        vectorizer = TfidfVectorizer(tokenizer=purpose_tokens, token_pattern=None, lowercase=False,
                                    stop_words=STOP_WORDS, ngram_range=ngrams, min_df=3,
                                    max_df=cutoff, max_features=maximum, sublinear_tf=True, dtype=np.float32)
        matrices.append(vectorizer.fit_transform(texts))
        vectorizers.append(vectorizer)
    return metadata, matrices, vectorizers


def combine_views(metadata, body, metadata_weight):
    if not 0 <= metadata_weight <= 1:
        raise ValueError("metadata_weight must be in [0, 1]")
    # Each TF-IDF row has unit norm: squared weights control contributions to
    # cosine similarity before dimensionality reduction.
    return sparse.hstack([metadata * np.sqrt(metadata_weight), body * np.sqrt(1 - metadata_weight)], format="csr")


def embedding(matrix, seed, dimensions=128):
    rank = min(dimensions, matrix.shape[0] - 1, matrix.shape[1] - 1)
    if rank < 2:
        raise ValueError("Too little data for a purpose embedding")
    model = TruncatedSVD(n_components=rank, n_iter=7, random_state=seed)
    return normalize(model.fit_transform(matrix)), float(model.explained_variance_ratio_.sum())


def fit_clusters(vectors, count, seed):
    return KMeans(n_clusters=count, n_init=10, max_iter=300, random_state=seed).fit(vectors)


def aligned_labels(reference, alternate):
    """Maximum-overlap label alignment for partition comparison, not validation."""
    overlap = contingency_matrix(reference, alternate)
    rows, columns = linear_sum_assignment(-overlap)
    ref_ids, alt_ids = np.unique(reference), np.unique(alternate)
    mapping = {int(alt_ids[c]): int(ref_ids[r]) for r, c in zip(rows, columns)}
    return np.array([mapping[int(value)] for value in alternate])


def cluster_terms(matrix, vectorizer, labels, cluster, count=12):
    overall = np.asarray(matrix.mean(axis=0)).ravel()
    mean = np.asarray(matrix[labels == cluster].mean(axis=0)).ravel()
    # Contrastive TF-IDF means emphasize terms enriched in this cluster.
    scores = mean - overall
    vocabulary = vectorizer.get_feature_names_out()
    order = np.argsort(-scores, kind="stable")
    return [str(vocabulary[i]) for i in order[:count] if scores[i] > 0]


def assignment_margins(distances):
    nearest = np.partition(distances, 1, axis=1)[:, :2]
    nearest.sort(axis=1)
    return (nearest[:, 1] - nearest[:, 0]) / np.maximum(nearest[:, 1], 1e-12)

"""Custom MinHash, banded LSH, and exact word-shingle Jaccard verification."""
from __future__ import annotations

from collections import defaultdict
import hashlib
from itertools import combinations
from typing import Iterable

import numpy as np

# A prime below 2**32 keeps a*x+b strictly below 2**64 without overflow.
PRIME = 4294967291


def shingles(tokens: tuple[str, ...], width: int = 5) -> frozenset[str]:
    if width < 1:
        raise ValueError("Shingle width must be positive")
    return frozenset("\x1f".join(tokens[i:i + width])
                     for i in range(len(tokens) - width + 1))


def jaccard(left: frozenset[str], right: frozenset[str]) -> float:
    """Exact similarity over strings; empty documents never count as matches."""
    if not left or not right:
        return 0.0
    intersection = len(left & right)
    return intersection / (len(left) + len(right) - intersection)


def signatures(documents: list[frozenset[str]], permutations: int = 128,
               seed: int = 42) -> np.ndarray:
    if permutations < 1:
        raise ValueError("permutations must be positive")
    rng = np.random.default_rng(seed)
    a = rng.integers(1, PRIME, size=permutations, dtype=np.uint64)
    b = rng.integers(0, PRIME, size=permutations, dtype=np.uint64)
    result = np.full((len(documents), permutations), PRIME, dtype=np.uint64)
    for i, document in enumerate(documents):
        hashes = np.fromiter((int.from_bytes(hashlib.blake2b(s.encode(), digest_size=4).digest(),
                                             "little") % PRIME for s in document), dtype=np.uint64)
        # Chunking bounds the temporary matrix independently of document length.
        for start in range(0, len(hashes), 1024):
            values = (hashes[start:start + 1024, None] * a + b) % PRIME
            result[i] = np.minimum(result[i], values.min(axis=0))
    return result


def lsh_candidates(matrix: np.ndarray, bands: int = 32, rows: int = 4,
                   max_candidates: int = 2_000_000) -> set[tuple[int, int]]:
    if bands < 1 or rows < 1 or bands * rows != matrix.shape[1]:
        raise ValueError("bands * rows must equal the signature length")
    if max_candidates < 1:
        raise ValueError("max_candidates must be positive")
    pairs: set[tuple[int, int]] = set()
    valid = np.any(matrix != PRIME, axis=1)
    for band in range(bands):
        buckets = defaultdict(list)
        for i in np.flatnonzero(valid):
            key = matrix[i, band * rows:(band + 1) * rows].tobytes()
            buckets[key].append(int(i))
        for bucket in buckets.values():
            for pair in combinations(bucket, 2):
                pairs.add(pair)
                if len(pairs) > max_candidates:
                    raise RuntimeError("Candidate cap exceeded; use a smaller sample or more rows per band. "
                                       "No candidates have been silently discarded.")
    return pairs


def verified_edges(documents: list[frozenset[str]], pairs: Iterable[tuple[int, int]],
                   threshold: float) -> dict[tuple[int, int], float]:
    if not 0 < threshold <= 1:
        raise ValueError("threshold must be in (0, 1]")
    edges = {}
    for i, j in pairs:
        smaller, larger = sorted((len(documents[i]), len(documents[j])))
        if not larger or smaller / larger < threshold:
            continue  # A rigorous upper bound, not an approximation.
        score = jaccard(documents[i], documents[j])
        if score >= threshold:
            edges[(i, j)] = score
    return edges


def retrieval_metrics(truth: set, candidates: set, selected: set, possible: int) -> dict:
    true_positive = len(truth & selected)
    return {
        "possible_pairs": possible, "candidate_pairs": len(candidates),
        "true_pairs": len(truth), "retrieved_true_pairs": true_positive,
        "missed_true_pairs": len(truth - selected),
        "candidate_precision": len(truth & candidates) / len(candidates) if candidates else None,
        "verified_precision": true_positive / len(selected) if selected else None,
        "recall": true_positive / len(truth) if truth else None,
        "candidate_reduction": 1 - len(candidates) / possible if possible else None,
    }
